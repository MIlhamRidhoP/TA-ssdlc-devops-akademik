18 | const kuotaDariIps = (ips) => {
19 |   if (ips === null) return 19;
20 |   if (ips >= 3) return 24;
21 |   if (ips >= 2.5) return 21;
22 |   if (ips >= 2) return 18;
23 |   return 15;

26 | // IPS dari semester terakhir sebelum `semester` yang punya nilai published
27 | const ipsSebelum = (grades, semester) => {
28 |   const batas = semesterKey(semester);
29 |   const perSemester = new Map();
30 | 
31 |   for (const g of grades) {
32 |     if (semesterKey(g.semester) >= batas) continue;
33 |     const s = perSemester.get(g.semester) || { bobot: 0, sks: 0 };
34 |     s.bobot += BOBOT[g.nilai_huruf] * g.sks;
35 |     s.sks += g.sks;
36 |     perSemester.set(g.semester, s);
37 |   }
38 | 
39 |   if (perSemester.size === 0) return null;
40 | 
41 |   const terakhir = [...perSemester.keys()].sort((a, b) => semesterKey(b) - semesterKey(a))[0];
42 |   const { bobot, sks } = perSemester.get(terakhir);
43 |   return { semester: terakhir, ips: Math.round((bobot / sks) * 100) / 100 };
44 | };

2. Report one entry per distinct issue per location. If the same flaw appears in several functions, report each location as a separate entry. Do not report the same issue at the same location twice.
3. "file": the path exactly as written in the FILE header.
4. "baris": the line number, taken from the listing, of the line most directly responsible for the issue (usually the sink). Use 0 only if no single line applies.
5. "fungsi": the name of the enclosing function exactly as written in the code. For an issue in a route definition, write the HTTP method and path, for example "PUT /:id/status". Use an empty string if neither applies.
6. "jenis": a short name for the issue type, for example "SQL Injection".
7. "cwe": the single most specific CWE identifier, in the form "CWE-89".
8. "kategori_owasp": one code from the OWASP Top 10:2025 list below. Use "Other" if none fits.
9. "penjelasan": two to four sentences in Bahasa Indonesia. State where the input comes from, what reaches the sensitive operation, and what protection is missing. Keep code identifiers, file paths, and CWE identifiers unchanged. Give a concise explanation, not a step-by-step account of your reasoning.
10. "sifat": "vulnerability" if a party without the corresponding right can concretely exploit it in the current code and configuration; "hardening" if it is a missing best practice with no direct exploitation path in the current code (for example a missing security header or missing rate limiting).
11. "keyakinan":
    - "high": the full path from input to the sensitive operation is visible in the provided code and no effective protection exists along it.
    - "medium": the path is only partly visible, or the conclusion depends on an assumption about behavior that is not fully shown.
    - "low": the code matches a suspicious pattern, but no clear exploitation path is visible.
12. If you find no issues, return {"temuan": []}.
